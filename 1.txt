<script>
const STORAGE_CONFIG = 'album_global_config'
let globalConfig = null
let currentPath = 'public/albums'
let viewMode = 'icon'
let showDatePrefix = false
let copyCache = { path: '', name: '', type: '', action: '' }
let selectedItem = null
const API_BASE = 'https://api.github.com'

// ========== 配置初始化 ==========
function checkConfig() {
  globalConfig = JSON.parse(localStorage.getItem(STORAGE_CONFIG))
  if (!globalConfig) {
    alert("请先完成系统配置！")
    return false
  }
  return true
}
function parseRepo() {
  const url = globalConfig.repoUrl
  const arr = url.replace('https://github.com/', '').split('/')
  return { owner: arr[0], repo: arr[1] }
}

// ========== GitHub API ==========
async function githubApi(path, method = 'GET', body = null) {
  const { owner, repo } = parseRepo()
  const headers = {
    'Authorization': `token ${globalConfig.githubToken}`,
    'Accept': 'application/vnd.github.v3+json'
  }
  const res = await fetch(`${API_BASE}/repos/${owner}/${repo}/contents/${path}`, {
    method, headers, body: body ? JSON.stringify(body) : null
  })
  if (!res.ok) throw new Error('请求失败')
  return res.json()
}

// ========== 工具函数 ==========
function getDatePrefix() {
  const d = new Date()
  const y = d.getFullYear()
  const m = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  return `${y}-${m}-${day}-`
}
function pureName(fullName) {
  const reg = /^\d{4}-\d{2}-\d{2}-/
  return reg.test(fullName) ? fullName.replace(reg, '') : fullName
}
function fullName(dispName) {
  const reg = /^\d{4}-\d{2}-\d{2}-/
  return reg.test(dispName) ? dispName : getDatePrefix() + dispName
}
function getFileIcon(type, name) {
  if (type === 'dir') return '📁'
  const ext = name.split('.').pop().toLowerCase()
  return ['jpg','jpeg','png','webp','gif'].includes(ext) ? '🖼️' : '📄'
}

// ========== 目录树 ==========
async function loadTree() {
  const tree = document.getElementById('treeView')
  tree.innerHTML = ''
  await scanDirTree('public/albums', tree, 0)
}
async function scanDirTree(path, parent, level) {
  try {
    const list = await githubApi(path)
    const dirs = list.filter(x => x.type === 'dir' && !x.name.startsWith('.'))
    for (const d of dirs) {
      const item = document.createElement('div')
      item.className = `tree-item ${level > 0 ? 'tree-indent' : ''}`
      item.dataset.path = d.path
      item.innerHTML = `<span>📁</span> ${showDatePrefix ? d.name : pureName(d.name)}`
      if (d.path === currentPath) item.classList.add('active')
      item.addEventListener('click', () => navigateTo(d.path))
      parent.appendChild(item)
      await scanDirTree(d.path, parent, level + 1)
    }
  } catch(e) { console.error(e) }
}

// ========== 面包屑导航 ==========
function renderBreadcrumb(path) {
  const bc = document.getElementById('breadcrumb')
  const parts = path.split('/')
  let temp = ''
  let html = `<a data-path="public/albums">相册根目录</a>`
  for (let i = 2; i < parts.length; i++) {
    temp += '/' + parts[i]
    const p = 'public/albums' + temp
    const n = showDatePrefix ? parts[i] : pureName(parts[i])
    html += ` / <a data-path="${p}">${n}</a>`
  }
  bc.innerHTML = html
  bc.querySelectorAll('a').forEach(a => {
    a.addEventListener('click', () => navigateTo(a.dataset.path))
  })
}

// ========== 文件列表渲染 ==========
async function renderFileList(path) {
  const area = document.getElementById('fileArea')
  area.innerHTML = ''
  try {
    const list = await githubApi(path)
    const dirs = list.filter(x => x.type === 'dir' && !x.name.startsWith('.'))
    const files = list.filter(x => x.type === 'file' && !x.name.startsWith('.'))
    const all = [...dirs, ...files]

    if (viewMode === 'icon') {
      area.className = 'file-area view-icon'
      all.forEach(item => {
        const card = document.createElement('div')
        card.className = 'file-card'
        card.dataset.path = item.path
        card.dataset.type = item.type
        card.dataset.sha = item.sha
        const dispName = showDatePrefix ? item.name : pureName(item.name)
        card.innerHTML = `
          <div class="file-icon">${getFileIcon(item.type, item.name)}</div>
          <div class="file-name">${dispName}</div>
        `
        card.addEventListener('click', e => {
          e.stopPropagation()
          clearSelect()
          card.classList.add('selected')
          selectedItem = item
        })
        card.addEventListener('dblclick', () => {
          if (item.type === 'dir') navigateTo(item.path)
          else openPreview(item)
        })
        card.addEventListener('contextmenu', e => {
          e.preventDefault()
          clearSelect()
          card.classList.add('selected')
          selectedItem = item
          showContextMenu(e.clientX, e.clientY, true)
        })
        area.appendChild(card)
      })
    } else {
      area.className = 'file-area'
      const table = document.createElement('table')
      table.className = 'view-list'
      table.innerHTML = `
        <thead>
          <tr><th>名称</th><th>类型</th><th>大小</th><th>路径</th></tr>
        </thead>
        <tbody id="listBody"></tbody>
      `
      area.appendChild(table)
      const tbody = document.getElementById('listBody')
      all.forEach(item => {
        const tr = document.createElement('tr')
        tr.dataset.path = item.path
        tr.dataset.type = item.type
        tr.dataset.sha = item.sha
        const dispName = showDatePrefix ? item.name : pureName(item.name)
        const typeText = item.type === 'dir' ? '文件夹' : '文件'
        const sizeText = item.type === 'dir' ? '-' : (item.size / 1024).toFixed(2) + ' KB'
        tr.innerHTML = `
          <td>${getFileIcon(item.type, item.name)} ${dispName}</td>
          <td>${typeText}</td>
          <td>${sizeText}</td>
          <td>${item.path}</td>
        `
        tr.addEventListener('click', e => {
          e.stopPropagation()
          clearSelect()
          tr.classList.add('selected')
          selectedItem = item
        })
        tr.addEventListener('dblclick', () => {
          if (item.type === 'dir') navigateTo(item.path)
          else openPreview(item)
        })
        tr.addEventListener('contextmenu', e => {
          e.preventDefault()
          clearSelect()
          tr.classList.add('selected')
          selectedItem = item
          showContextMenu(e.clientX, e.clientY, true)
        })
        tbody.appendChild(tr)
      })
    }
  } catch(e) {
    area.innerHTML = '<p>加载目录失败</p>'
  }
}

// 清空选中
function clearSelect() {
  document.querySelectorAll('.selected').forEach(el => el.classList.remove('selected'))
  selectedItem = null
}

async function navigateTo(path) {
  currentPath = path
  renderBreadcrumb(path)
  await loadTree()
  await renderFileList(path)
  hideContextMenu()
}

// 显示菜单：hasItem = true 选中文件 / false 空白区域
function showContextMenu(x, y, hasItem) {
  // 切换菜单项可用状态
  document.querySelectorAll('.item-only').forEach(el => {
    el.classList.toggle('menu-disabled', !hasItem)
  })
  // 粘贴项：有复制/剪切内容才可用
  const pasteItem = contextMenu.querySelector('[data-action="paste"]')
  pasteItem.classList.toggle('menu-disabled', !copyCache.path)

  contextMenu.style.left = x + 'px'
  contextMenu.style.top = y + 'px'
  contextMenu.style.display = 'block'
}
function hideContextMenu() {
  contextMenu.style.display = 'none'
}

// 右键菜单动作
contextMenu.querySelectorAll('.menu-item:not(.menu-disabled)').forEach(item => {
  item.addEventListener('click', async () => {
    hideContextMenu()
    const act = item.dataset.action

    switch(act) {
      case 'newFolder':
        document.getElementById('newFolderModal').style.display = 'flex'
        document.getElementById('newFolderName').value = ''
        document.getElementById('newFolderName').focus()
        break
      case 'upload':
        document.getElementById('fileUpload').click()
        break
      case 'paste':
        await doPaste()
        break
      case 'preview':
        if(selectedItem) openPreview(selectedItem)
        break
      case 'copy':
        if(selectedItem) {
          copyCache = { path: selectedItem.path, name: selectedItem.name, type: selectedItem.type, action: 'copy' }
          alert('已复制，在空白处右键选择【粘贴】')
        }
        break
      case 'cut':
        if(selectedItem) {
          copyCache = { path: selectedItem.path, name: selectedItem.name, type: selectedItem.type, action: 'cut' }
          alert('已剪切，在空白处右键选择【粘贴】')
        }
        break
      case 'rename':
        if(selectedItem) openRename()
        break
      case 'delete':
        if(selectedItem) await doDelete()
        break
      case 'prop':
        if(selectedItem) openProp()
        break
    }
  })
})

// 带完整进度的文件上传
document.getElementById('fileUpload').addEventListener('change', async e => {
  const files = e.target.files
  if (!files.length) return
  uploadPanel.classList.remove('hidden')

  for (const file of files) {
    // 单文件进度项
    const item = document.createElement('div')
    item.className = 'upload-item'
    item.innerHTML = `
      <div class="upload-filename">${file.name}</div>
      <div class="progress-bar"><div class="progress-inner"></div></div>
      <div class="upload-status">准备上传...</div>
    `
    uploadList.appendChild(item)
    const progressInner = item.querySelector('.progress-inner')
    const statusText = item.querySelector('.upload-status')

    try {
      statusText.textContent = '读取文件中...'
      // 分段模拟真实进度 + FileReader 读取进度
      const reader = new FileReader()
      reader.onprogress = (ev) => {
        if (ev.lengthComputable) {
          const percent = Math.round((ev.loaded / ev.total) * 40)
          progressInner.style.width = percent + '%'
          statusText.textContent = `读取文件 ${ev.loaded}/${ev.total} 字节`
        }
      }

      reader.onload = async () => {
        progressInner.style.width = '45%'
        statusText.textContent = '转换Base64并提交请求...'
        const b64 = reader.result.split(',')[1]
        const targetPath = `${currentPath}/${file.name}`

        // 模拟网络请求进度
        let reqProgress = 45
        const progressTimer = setInterval(() => {
          reqProgress += Math.floor(Math.random() * 3)
          if (reqProgress >= 90) {
            reqProgress = 90
            clearInterval(progressTimer)
          }
          progressInner.style.width = reqProgress + '%'
          statusText.textContent = `请求上传中 ${reqProgress}%`
        }, 80)

        await githubApi(targetPath, 'PUT', { message: `上传 ${file.name}`, content: b64 })
        clearInterval(progressTimer)
        progressInner.style.width = '100%'
        statusText.textContent = '上传完成'
        statusText.className = 'upload-status status-success'
      }

      reader.onerror = () => {
        progressInner.style.width = '0%'
        statusText.textContent = '文件读取失败'
        statusText.className = 'upload-status status-error'
      }
      reader.readAsDataURL(file)
    } catch (err) {
      progressInner.style.width = '0%'
      statusText.textContent = '上传失败：' + err.message
      statusText.className = 'upload-status status-error'
    }
  }
  // 上传完毕刷新列表
  setTimeout(() => renderFileList(currentPath), 1500)
  // 清空文件选择，支持重复选同一批文件
  e.target.value = ''
})
